# 数据集批量执行方案（完善版）

## 一、系统概述

本方案用于批量处理多个测试案例（case），自动执行检查点提取（步骤1）和AI验证（步骤2），每次批量执行都会重新生成最新的步骤1输出JSON文件。

### 1.1 核心逻辑

**关键特性：**
- ✅ 每次批量执行都会执行步骤1，生成新的步骤1输出JSON文件
- ✅ 步骤1生成失败时，不会执行步骤2
- ✅ 步骤2始终使用当前批次生成的最新JSON文件
- ✅ 串行处理，按顺序逐个执行case
- ✅ 完整的进度显示和错误处理

### 1.2 执行流程

```
┌─────────────────────────────────────────────────────┐
│              单个Case执行流程                          │
└─────────────────────────────────────────────────────┘

                    │
                    ▼
        ┌───────────────────────────┐
        │  1. 验证case结构            │
        │     检查必需文件是否存在    │
        └───────────────────────────┘
                    │
                    ▼
        ┌───────────────────────────┐
        │  2. 执行步骤1（必须执行）    │
        │     调用API提取检查点       │
        │     生成新的JSON文件        │
        │     文件名带当前时间戳      │
        └───────────────────────────┘
                    │
        ┌───────────┴───────────┐
        │                       │
        ▼                       ▼
   步骤1成功               步骤1失败
        │                       │
        ▼                       ▼
 ┌─────────────────┐      直接返回失败
 │  3. 执行步骤2    │         │
 │  使用本次生成的  │         │
 │  JSON文件       │         │
 └─────────────────┘         │
        │                    │
        ▼                    │
   生成验证报告               │
        │                    │
        └────────┬───────────┘
                 ▼
          返回处理结果
```

## 二、数据集结构规范

### 2.1 Case文件夹结构

```
datasets/
├── case_001/
│   ├── bid_invitation.pdf          # 招标文件（必需，支持PDF/DOCX）
│   ├── bid_proposal.pdf             # 投标文件（必需，支持PDF/DOCX）
│   ├── manual_checkpoints.xlsx    # 人工标注检查点（必需）
│   ├── case_001_check_point_20260203_161153.json  # 步骤1输出（自动生成）
│   └── case_001_validation_report_20260203_161822.md  # 步骤2输出（自动生成）
├── case_002/
│   ├── bid_invitation.docx         # 支持DOCX格式
│   ├── bid_proposal.docx
│   ├── manual_checkpoints.xlsx
│   ├── case_002_check_point_YYYYMMDD_HHMMSS.json  # 最新生成的JSON
│   └── case_002_validation_report_YYYYMMDD_HHMMSS.md
├── case_003/
│   ├── 招标文件.pdf                 # 支持中文命名
│   ├── 投标文件.pdf
│   ├── 人工标注.xlsx
│   └── ...
└── ...
```

**支持的文件格式**：

| 文件类型 | 支持格式 | 说明 |
|---------|---------|------|
| 招标文件 | `.pdf`, `.docx` | 必需，支持中英文命名 |
| 投标文件 | `.pdf`, `.docx` | 必需，支持中英文命名 |
| 标注文件 | `.xlsx`, `.xls` | 必需，人工标注的检查点 |

**文件识别规则**：

| 文件类型 | 识别优先级 |
|---------|-----------|
| 招标文件 | `*招标*.pdf/.docx` → `*zb*.pdf/.docx` → 第1个`.pdf/.docx` |
| 投标文件 | `*投标*.pdf/.docx` → `*tb*.pdf/.docx` → 第2个`.pdf/.docx` |
| 标注文件 | 第1个`.xlsx`或`.xls`文件 |

### 2.2 文件命名规范

**步骤1输出JSON文件：**
- 格式：`{case_name}_check_point_{timestamp}.json`
- 示例：`case_001_check_point_20260203_161153.json`

**步骤2输出报告文件：**
- 格式：`{case_name}_validation_report_{timestamp}.md`
- 示例：`case_001_validation_report_20260203_161822.md`

```python
                self.latest_checkpoint_json = json_files[0]

                bid_files_found.append(f"步骤1输出文件: {self.latest_checkpoint_json.name}")

                missing_files.append("招标文件")
            if not self.bid_proposal:
                missing_files.append("投标文件")
            if not self.manual_checkpoint:
                missing_files.append("人工标注文件")
```

### 3.2 批量处理主程序

```python
import argparse
import json
import logging
from pathlib import Path
from datetime import datetime
from typing import List, Dict, Any
# 注意：串行处理，按顺序逐个执行case
from tqdm import tqdm

# 配置日志
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
    handlers=[
        logging.FileHandler('batch_processing.log', encoding='utf-8'),
        logging.StreamHandler()
    ]
)
logger = logging.getLogger(__name__)


class BatchProcessor:
    """批量处理器"""

    def __init__(self, dataset_path: str, config: dict = None):
        self.dataset_path = Path(dataset_path)
        self.config = config or {}
        self.logger = logging.getLogger("BatchProcessor")

    def discover_cases(self) -> List[Path]:
        """
        扫描数据集目录，发现所有case文件夹

        Returns:
            case文件夹路径列表
        """
        cases = []

        if not self.dataset_path.exists():
            self.logger.error(f"数据集目录不存在: {self.dataset_path}")
            return cases

        # 遍历数据集目录
        for item in self.dataset_path.iterdir():
            if item.is_dir() and not item.name.startswith('_'):
                cases.append(item)

        cases.sort()  # 按名称排序
        self.logger.info(f"✅ 发现 {len(cases)} 个case文件夹")
        return cases

    def process_single_case(self, case_path: Path) -> Dict[str, Any]:
        """
        处理单个case

        Args:
            case_path: case文件夹路径

        Returns:
            处理结果字典
        """
        from case_processor import CaseProcessor  # 导入上面的CaseProcessor

        processor = CaseProcessor(str(case_path), self.config)
        return processor.process()

    def run_batch_processing(
        self,
        single_case: str = None
    ) -> Dict[str, Any]:
        """
        运行批量处理

        执行流程：
        - 按顺序逐个执行case（串行处理）
        - 每个case都会执行步骤1生成新的JSON文件
        - 步骤1成功后才执行步骤2

        Args:
            single_case: 只处理指定的case（可选）

        Returns:
            批量处理汇总结果
        """
        # 发现所有case
        all_cases = self.discover_cases()

        if not all_cases:
            self.logger.error("未发现任何case文件夹")
            return {"status": "failed", "error": "未发现case"}

        # 过滤单个case（如果指定）
        if single_case:
            all_cases = [c for c in all_cases if c.name == single_case]
            if not all_cases:
                self.logger.error(f"未找到指定的case: {single_case}")
                return {"status": "failed", "error": f"未找到case: {single_case}"}
            self.logger.info(f"🎯 仅处理单个case: {single_case}")

        # 结果统计
        results = {
            "total": len(all_cases),
            "success": 0,
            "failed": 0,
            "skipped": 0,
            "cases": [],
            "start_time": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
            "end_time": None
        }

        # 串行处理（按顺序逐个执行）
        self.logger.info(f"🚀 开始批量处理，按顺序逐个执行")
        self.logger.info(f"{'='*80}")

        for case_path in all_cases:
            case_name = case_path.name
            self.logger.info(f"\n正在处理: {case_name} ({i+1}/{len(all_cases)})")

            # 处理单个case
            result = self.process_single_case(case_path)

            # 记录结果
            results["cases"].append(result)

            # 更新统计
            if result["status"] == "success":
                results["success"] += 1
                self.logger.info(f"✅ {case_name} 处理成功")
            else:
                results["failed"] += 1
                self.logger.error(f"❌ {case_name} 失败: {result.get('error')}")

        results["end_time"] = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

        # 打印汇总
        self.logger.info(f"{'='*80}")
        self.logger.info(f"📊 批量处理完成！")
        self.logger.info(f"总计: {results['total']} | 成功: {results['success']} | 失败: {results['failed']}")
        self.logger.info(f"{'='*80}")

        return results

    def save_results(self, results: Dict[str, Any]) -> Path:
        """
        保存批量处理结果

        Args:
            results: 处理结果字典

        Returns:
            保存的结果文件路径
        """
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        output_file = self.dataset_path / f"batch_results_{timestamp}.json"

        with open(output_file, 'w', encoding='utf-8') as f:
            json.dump(results, f, ensure_ascii=False, indent=2)

        self.logger.info(f"💾 批量处理结果已保存: {output_file}")
        return output_file


def main():
    """主函数"""
    parser = argparse.ArgumentParser(
        description="招标文件检查点批量验证工具",
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog="""
示例用法:
  # 基本用法
  python main.py --dataset ./datasets

  # 只处理单个case
  python main.py --dataset ./datasets --single-case case_001

注意:
  - 每次批量执行都会重新执行步骤1，生成新的JSON文件
  - 按顺序逐个执行case（串行处理）
        """
    )

    parser.add_argument(
        '--dataset',
        type=str,
        required=True,
        help='数据集根目录路径'
    )

    parser.add_argument(
        '--single-case',
        type=str,
        help='只处理指定的case'
    )

    args = parser.parse_args()

    # 创建批量处理器
    processor = BatchProcessor(args.dataset)

    # 执行批量处理
    results = processor.run_batch_processing(
        single_case=args.single_case
    )

    # 保存结果
    processor.save_results(results)

    # 根据结果返回退出码
    if results["failed"] > 0:
        exit(1)
    exit(0)


if __name__ == "__main__":
    main()
```

## 四、关键功能说明

### 4.1 步骤1执行逻辑

**核心原则：每次批量执行都会执行步骤1，生成新的JSON文件**

**实现位置：**
- `extract_checkpoints()`: 执行步骤1，生成新的JSON文件
- `process()`: 主处理流程，先执行步骤1，成功后再执行步骤2

**逻辑流程：**
```
1. 验证case结构（检查必需文件）
2. 执行步骤1：调用API提取检查点
3. 生成新的JSON文件（带当前时间戳）
4. 如果步骤1成功 → 执行步骤2
5. 如果步骤1失败 → 直接返回失败，不执行步骤2
```

### 3.4 与现有系统集成

```python
"""
将批量处理器集成到现有的pytest测试流程中
"""
from api_workflow import PDFExtractionService  # 假设的现有提取服务
from llm_matcher import LLMMatcherService  # 假设的现有匹配服务


class IntegratedCaseProcessor(CaseProcessor):
    """集成现有API的Case处理器"""

    def __init__(self, case_path: str, config):
        super().__init__(case_path, config)

        # 初始化现有服务
        self.pdf_extraction_service = PDFExtractionService()
        self.llm_matcher_service = LLMMatcherService(api_key=config.get('ZHIPUAI_API_KEY'))

    def extract_checkpoints(self) -> Optional[Path]:
        """执行步骤1: 调用现有API提取检查点（每次批量执行都会执行）"""
        try:
            # 调用现有的提取API
            self.logger.info("🚀 调用API提取检查点...")

            # 调用现有服务
            extraction_result = self.pdf_extraction_service.extract_checkpoints_from_pdfs(
                zb_pdf_path=str(self.bid_invitation),
                tb_pdf_path=str(self.bid_proposal)
            )

            # 生成输出文件（带当前时间戳）
            timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
            output_filename = f"{self.case_name}_check_point_{timestamp}.json"
            output_path = self.case_path / output_filename

            # 保存提取结果
            with open(output_path, 'w', encoding='utf-8') as f:
                json.dump(extraction_result, f, ensure_ascii=False, indent=2)

            self.latest_checkpoint_json = output_path
            self.logger.info(f"✅ 步骤1完成，生成文件: {output_filename}")
            return output_path

        except Exception as e:
            self.logger.error(f"❌ 步骤1提取失败: {e}")
            return None

    def validate_and_generate_report(self, checkpoint_json: Optional[Path] = None) -> Optional[Path]:
        """执行步骤2: 调用现有LLM匹配服务"""
        try:
            if checkpoint_json is None:
                checkpoint_json = self.latest_checkpoint_json

            if checkpoint_json is None:
                self.logger.error("❌ 未找到步骤1输出JSON文件")
                return None

            self.logger.info("🚀 调用AI进行语义匹配...")

            # 调用现有LLM匹配服务
            result_path = self.llm_matcher_service.run_matching_task(
                excel_path=str(self.manual_checkpoint),
                json_path=str(checkpoint_json)
            )

            self.logger.info(f"✅ 步骤2完成，生成报告: {result_path}")
            return Path(result_path)

        except Exception as e:
            self.logger.error(f"❌ 步骤2验证失败: {e}")
            return None
```
